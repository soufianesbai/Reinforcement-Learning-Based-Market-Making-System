% --- Slide 3 : Introduction & contexte
\begin{frame}{Introduction \& contexte}
    \begin{block}{Constat}
        Les marchés financiers modernes reposent sur des \alert{moteurs de carnet d'ordres}
        capables de traiter des centaines de milliers d'ordres par seconde,
        avec des latences de l'ordre de la microseconde.
    \end{block}

    \vspace{0.3cm}
    \textbf{Notre projet réunit deux briques complémentaires :}
    \begin{itemize}
        \item une \alert{infrastructure de marché} (matching engine + ingestion temps réel) ;
        \item une \alert{collection d'agents de trading} algorithmiques, depuis les
              heuristiques classiques jusqu'aux méthodes d'apprentissage par renforcement modernes.
    \end{itemize}

    \vspace{0.3cm}
    \begin{block}{Objectif}
        Construire une plateforme de bout en bout permettant de \textbf{simuler},
        \textbf{entraîner} et \textbf{évaluer} des stratégies de trading sur des
        données crypto réelles (Binance).
    \end{block}
\end{frame}

% --- Slide 9 : Structures de données
\begin{frame}{Moteur C++ -- Structures de données}
    \begin{columns}[T]
        \column{0.55\textwidth}
        \textbf{Représentation du carnet}
        \begin{itemize}
            \item Deux \texttt{std::map} séparées
              \begin{itemize}
                \item bids triés \alert{décroissants}
                \item asks triés \alert{croissants}
              \end{itemize}
            \item Niveau de prix : \texttt{std::list<OrderPointer>} (FIFO)
            \item \texttt{std::unordered\_map<OrderId, OrderEntry>} avec itérateur stocké
        \end{itemize}

        \column{0.45\textwidth}
        \textbf{Choix justifiés}
        \begin{itemize}
            \item Maps séparées $\Rightarrow$ deux ordres de tri opposés.
            \item Itérateur stocké $\Rightarrow$ \alert{annulation $O(1)$}.
            \item \texttt{shared\_ptr} : références multiples sans copie.
            \item \texttt{Market} convertis en limit agressifs.
        \end{itemize}
    \end{columns}

    \vspace{0.2cm}
    \begin{center}
    \footnotesize
    \begin{tabular}{lll}
        \toprule
        \textbf{Opération} & \textbf{Complexité} & \textbf{Justification} \\
        \midrule
        Ajout d'ordre   & $O(\log n)$   & insertion dans la map \\
        Annulation      & $O(1)$        & itérateur stocké \\
        Modification    & $O(\log n)$   & cancel + insert \\
        Matching        & $O(k \log n)$ & $k$ trades exécutés \\
        \bottomrule
    \end{tabular}
    \end{center}
\end{frame}

% --- Slide 10 : Algorithme de matching
\begin{frame}{Algorithme de matching}
    \begin{block}{Condition de match}
        Un ordre s'exécute dès que : \quad
        $\text{prix}_{\text{buy}} \geq \text{best ask} \quad$ ou $\quad
        \text{prix}_{\text{sell}} \leq \text{best bid}$
    \end{block}

    \vspace{0.2cm}
    \textbf{Priorité prix-temps}
    \begin{enumerate}
        \item \alert{Meilleur prix} d'abord (best bid pour les vendeurs, best ask pour les acheteurs).
        \item Au sein d'un niveau : \alert{ordre d'arrivée} (FIFO).
        \item Exécutions partielles autorisées (sauf \texttt{FillOrKill}).
    \end{enumerate}

    \vspace{0.3cm}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Bids (achats)}
        \begin{center}
        \scriptsize
        \begin{tabular}{cc}
            \toprule
            Prix & Quantité \\
            \midrule
            \alert{103.25} & 1234.56 \\
            103.24 & 891.23 \\
            103.23 & 2145.67 \\
            \bottomrule
        \end{tabular}
        \end{center}

        \column{0.5\textwidth}
        \textbf{Asks (ventes)}
        \begin{center}
        \scriptsize
        \begin{tabular}{cc}
            \toprule
            Prix & Quantité \\
            \midrule
            \alert{103.27} & 1567.89 \\
            103.28 & 723.45 \\
            103.29 & 1234.56 \\
            \bottomrule
        \end{tabular}
        \end{center}
    \end{columns}

    \begin{center}
        Spread : $103.27 - 103.25 = 0.02 \approx 1.94$ bps
    \end{center}
\end{frame}

% --- Slide 14 : Stratégies par règles
\begin{frame}{Stratégies par règles}
    \begin{block}{Mean Reversion (baseline interprétable)}
        \begin{itemize}
            \item ACHETER si prix $< \text{SMA} - 2\%$
            \item VENDRE si prix $> \text{SMA} + 2\%$
            \item ATTENDRE sinon
        \end{itemize}
    \end{block}

    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{RSI (14 périodes)}
        \begin{itemize}
            \item RSI $< 30$ : zone de survente $\Rightarrow$ \alert{ACHAT}
            \item RSI $> 70$ : zone de surachat $\Rightarrow$ \alert{VENTE}
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Bollinger Bands ($\pm 2\sigma$)}
        \begin{itemize}
            \item Sortie sous la bande inférieure $\Rightarrow$ \alert{ACHAT}
            \item Sortie au-dessus de la bande supérieure $\Rightarrow$ \alert{VENTE}
        \end{itemize}
    \end{columns}

    \vspace{0.3cm}
    \begin{block}{Architecture modulaire}
        Toutes les stratégies implémentent l'interface \texttt{BaseStrategy},
        sélectionnables via la variable d'environnement \texttt{STRATEGY=...}.
    \end{block}

% --- Slide 18 : DQN
\begin{frame}{DQN -- Deep Q-Network}
    \begin{block}{Architecture}
        Réseau \alert{dueling} avec activations SiLU et LayerNorm,
        \alert{Double DQN} pour atténuer la sur-estimation des Q-valeurs.
    \end{block}

    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Améliorations clés}
        \begin{itemize}
            \item \textit{Dueling} : sépare $V(s)$ et $A(s,a)$
            \item \textit{Double DQN} : cible découplée
            \item \textit{Replay buffer} 200\,k transitions
            \item Perte \alert{Smooth-L1} (Huber)
            \item \textit{Gradient clipping}
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Hyperparamètres}
        \begin{itemize}
            \item $\gamma = 0.995$
            \item lr $= 3 \times 10^{-4}$ (Adam)
            \item Batch $= 256$
            \item $\varepsilon$ : $1.0 \to 0.01$ (décroissant)
            \item Target network sync périodique
        \end{itemize}
    \end{columns}

    \vspace{0.2cm}
    \begin{block}{Optimisations}
        Détection automatique CUDA, lecture streaming des CSV (jusqu'à 3\,Go),
        sauvegarde périodique des poids.
    \end{block}
\end{frame}

% --- A2C : page texte
\begin{frame}{A2C -- Advantage Actor-Critic}
    \begin{block}{Principe}
        Combinaison \alert{acteur-critique} synchrone, mise à jour par avantage estimé.
    \end{block}

    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Architecture}
        \begin{itemize}
            \item \alert{Acteur} : politique $\pi_\theta(a|s)$
            \item \alert{Critique} : valeur $V_\phi(s)$
            \item Tronc commun partagé
            \item Mise à jour par \alert{avantage} : $A = R - V(s)$
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Pertes}
        \begin{align*}
            \mathcal{L}_\text{actor}   &= -\log \pi(a|s)\, A \\
            \mathcal{L}_\text{critic}  &= (R - V(s))^2 \\
            \mathcal{L}_\text{entropy} &= -\beta\, \mathcal{H}[\pi]
        \end{align*}
    \end{columns}
\end{frame}

% --- PPO : page texte
\begin{frame}{PPO -- Proximal Policy Optimization}
    \begin{block}{Idée clé : clipping du ratio de politique}
        \[
        \mathcal{L}^{\text{CLIP}} = \mathbb{E}_t\!\left[
            \min\!\big(r_t A_t,\;
            \mathrm{clip}(r_t, 1-\epsilon, 1+\epsilon)\, A_t\big)
        \right],\;
        r_t = \dfrac{\pi_\theta(a_t|s_t)}{\pi_{\theta_\text{old}}(a_t|s_t)}
        \]
    \end{block}

    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Avantages sur A2C}
        \begin{itemize}
            \item Stabilité accrue
            \item Pas de divergence brutale de la politique
            \item Plusieurs époques sur le même batch
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Hyperparamètres}
        \begin{itemize}
            \item Clip $\epsilon = 0.2$
            \item GAE $\lambda = 0.95$
            \item lr $= 3 \times 10^{-4}$
            \item Coef entropie $= 0.01$
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Dreamer v3 -- Architecture model-based}
    \begin{block}{Principe : apprendre dans l'imagination}
        Un \alert{World Model} apprend la dynamique du marché ;
        un acteur-critique est ensuite entraîné par
        \alert{rollouts imaginés} dans ce modèle, sans interagir avec l'environnement.
    \end{block}

    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{World Model}
        \begin{itemize}
            \item Encodeur d'observations
            \item \alert{RSSM} (Recurrent State-Space Model)
              \begin{itemize}
                \item Latents catégoriels $32 \times 32$
                \item GRU pour la mémoire récurrente
              \end{itemize}
            \item Prédicteurs : reward, continuation, observation
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Acteur-Critique}
        \begin{itemize}
            \item Entraîné sur \alert{rollouts imaginés}
            \item Pas d'interaction avec l'environnement réel pendant la phase d'imagination
            \item \texttt{symlog} / \texttt{symexp} pour stabilité numérique
              des récompenses à grande dynamique
        \end{itemize}
    \end{columns}
\end{frame}
