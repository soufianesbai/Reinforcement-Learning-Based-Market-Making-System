% --- Slide 8 : Structures de données et complexité
\begin{frame}{Structures de données}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \footnotesize
        \begin{itemize}
            \item Deux \texttt{std::map} séparées (bids \alert{$\downarrow$}, asks \alert{$\uparrow$})
            \item Niveau de prix : \texttt{std::list} (FIFO)
            \item \texttt{unordered\_map<id, iter>} : annulation \alert{$O(1)$}
        \end{itemize}

        \column{0.5\textwidth}
        \scriptsize
        \begin{tabular}{lll}
            \toprule
            \textbf{Op.} & \textbf{Complexité} & \textbf{Débit} \\
            \midrule
            Ajout         & $O(\log n)$  & $\approx 400$\,k/s \\
            Annulation    & $O(1)$       & $\approx 2$\,M/s \\
            Modification  & $O(\log n)$  & $\approx 270$\,k/s \\
            Matching      & $O(k\log n)$ & $\approx 350$\,k/s \\
            \bottomrule
        \end{tabular}
    \end{columns}

    \vspace{0.3cm}
    \begin{block}{Performances mesurées}
        Latence \alert{2--4\,$\mu$s} sur matériel de développement standard.\\
        Cible $\geq 10^5$ ops/s : \textbf{atteinte}.
    \end{block}
\end{frame}

Q(s,a) \leftarrow r + \gamma \max_{a'} Q_{\theta^-}(s',a')

        \begin{itemize}
            \item Action \alert{continue} : $(b_s,\,a_s)\!\in\![0,\,0.5\%]^2$
            \item Pas de pari directionnel
            \item Capturer le spread des \alert{deux côtés}
        \end{itemize}

        \vspace{0.25cm}
        \textbf{\color{enseirbblue}Reward Avellaneda-Stoikov}
        \[
          r = \underbrace{\Delta\!mid \times qty}_{\text{spread capturé}}
            - \lambda\,\frac{I^2}{\sigma}
            - \lambda_{adv}\,\delta_{adv}
        \]

        \vspace{0.2cm}
        \footnotesize
        \begin{tabular}{ll}
            \alert{Twin Q-net} & anti-surestimation \\
            \alert{$\alpha=0.2$ fixe} & entropie contrôlée \\
            \alert{Buffer 500k} & off-policy stable \\
            \alert{$\tau=0.005$} & soft target update \\
        \end{tabular}

    \end{columns}
\end{frame}

% --- Slide 20 : PPO -- Politique, Loss, Reward
\begin{frame}{PPO -- Politique, Loss \& Reward}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{\color{enseirbblue}Politique continue (market making)}
        \begin{itemize}
            \item Action $\mathbf{a}=(a_\text{ask}, a_\text{bid})\in[-1,1]^2$
            \item Réseau acteur--critique \textbf{64--64} (séparé)
        \end{itemize}
        \[
        p_\text{ask} = p\Bigl(1+\tfrac{a_\text{ask}+1}{2}\,s_{\max}\Bigr)
        \]
        \textbf{\color{enseirbblue}Reward}
        \[
        r_t = \underbrace{\Delta\text{cash}_t}_{\text{spread capturé}}
              - \underbrace{\lambda\, i_t^2\, p_t}_{\text{pénalité inventaire}}
        \]

        \column{0.48\textwidth}
        \textbf{\color{enseirbblue}Loss totale}
        \[
        \mathcal{L} = \mathcal{L}_\pi + 0{,}5\,\mathcal{L}_V - 0{,}01\,H(\pi)
        \]
        \textit{Policy loss (clipped surrogate)} :
        \[
        \mathcal{L}_\pi = -\mathbb{E}\!\left[\min\!\bigl(
            r_t\hat{A}_t,\,
            \mathrm{clip}(r_t,1\!-\!\varepsilon,1\!+\!\varepsilon)\hat{A}_t
        \bigr)\right]
        \vspace{-0.3cm}
        \]
        \scriptsize
        \begin{tabular}{ll}
            \toprule
            \textbf{Variable} & \textbf{Définition} \\
            \midrule
            $p$ & mid-price courant \\
            $s_{\max}=0{,}5\%$ & spread maximum autorisé \\
            $i_t \in[-10,10]$ & inventaire (SOL détenus) \\
            $\lambda=5\!\times\!10^{-4}$ & coeff pénalité inventaire \\
            $r_t=\pi_\text{new}/\pi_\text{old}$ & ratio nouvelles/anciennes proba \\
            $\hat{A}_t$ & avantage GAE normalisé \\
            $\varepsilon=0{,}2$ & borne de clipping PPO \\
            $\gamma=0{,}99$ & facteur d'actualisation \\
            $H(\pi)$ & entropie de la politique \\
            \bottomrule
        \end{tabular}
    \end{columns}
\end{frame}
